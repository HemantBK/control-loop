Column meanings:

## Why the printed figures are low: cache reads are not in `input_tokens`

The agents' gpt-4o calls hit OpenAI's automatic prompt cache. Inspect 0.3.263 records
`input_tokens = prompt_tokens - cached_tokens` and puts the cached part in
`input_tokens_cache_read` (`inspect_ai/model/_openai.py`, ModelOutput usage). The .out files show
the same split, e.g. `I: 95,685, CR: 63,872, O: 18,410`. `spend()` prices only `input_tokens`, so
1,536,640 cached gpt-4o input tokens in Phase 3 and 146,304 in §16.12 were left out. gpt-4o-mini
(the judge) recorded no cache reads. No cache writes were recorded.

The cache reads are reported separately here and not silently discounted:
- Price table, from `phases/phase3_allocation_monitor/sweep.py` PRICE: OpenAI list, $ per 1M
  tokens: gpt-4o $2.50 input / $10.00 output; gpt-4o-mini $0.15 / $0.60.
- Cached input is not in the repo. OpenAI's list price for cached input is 50% of input: gpt-4o
  $1.25, gpt-4o-mini $0.075. This price was not checked against OpenAI's page in this session.
  The upper-bound column does not depend on it.

  round 0 only.

  in Anthropic's claude-api skill (cached 2026-10-06). It is not recorded in the repo.
- (i) Flat $0.15 per cell gives $1.20. (ii) Characters at 4 per token: about 6.6k input and 0.95k
  output tokens per cell, so about $0.023 per cell and $0.18 in all.
- **The flat $0.15 is the upper bound**, with one caveat. Sonnet 5.5 runs adaptive thinking by
  default, and `raw` holds only the visible text. Hidden thinking tokens, billed as output, are not
  in (ii). The flat figure stays above the true cost unless thinking exceeds about 13k tokens per

## Script (exactly as run)

```python
"""Hand reconciliation of 2026-10-20 spend from the logs (no API calls). Run from the repo root:
    uv run --locked python - < (this script, saved anywhere)
"""
import json
from collections import defaultdict
from pathlib import Path

from inspect_ai.log import read_eval_log

# $ per 1M tokens (input, output): phases/phase3_allocation_monitor/sweep.py PRICE (OpenAI list).
PRICE = {"gpt-4o": (2.5, 10.0), "gpt-4o-mini": (0.15, 0.6)}
# OpenAI list price for cached input (not in the repo): 50% of input.
CACHED = {"gpt-4o": 1.25, "gpt-4o-mini": 0.075}
SONNET55 = (2.0, 10.0)  # claude-sonnet-5-5 list, $ per 1M in/out (claude-api skill model table, cached 2026-10-06)
CHARS_PER_TOKEN = 4
P3 = Path("results/phase3_allocation_monitor")
P2 = Path("results/phase2_claim_monitor/late")


def group_of(p: Path) -> str:
    parts = p.parts
    if parts[1] == "phase3_allocation_monitor":
        if parts[2] == "check":
            return f"P3 check/{parts[3]}"
        return f"P3 {parts[2]}/{parts[3]}/{parts[4]}"  # ordering/kind/cell
    i = parts.index("late")
    rest = parts[i + 1:]
    if rest[0] == "baselines":
        return f"P2 late/baselines/{rest[1]}"
    return f"P2 late/{rest[0]}/{rest[1]}"


def mkey(model: str) -> str:
    return "gpt-4o-mini" if "mini" in model else "gpt-4o"


files = sorted(list(P3.rglob("*.eval")) + [p for d in ("greedy_two", "score_only", "baselines/733_claim_score_only")
                                            for p in (P2 / d).rglob("*.eval")])
tok = defaultdict(lambda: defaultdict(lambda: [0, 0, 0, 0]))  # group -> model -> [in, out, cache_read, cache_write]
eps = defaultdict(int)
nfiles = defaultdict(int)
status = defaultdict(set)
for p in files:
    log = read_eval_log(str(p), header_only=True)
    g = group_of(p)
    nfiles[g] += 1
    status[g].add(log.status)
    eps[g] += (log.results.total_samples if log.results else 0)
    for model, u in (log.stats.model_usage or {}).items():
        t = tok[g][model]
        t[0] += u.input_tokens; t[1] += u.output_tokens
        t[2] += u.input_tokens_cache_read or 0; t[3] += u.input_tokens_cache_write or 0


def cost(model, t):
    pin, pout = PRICE[mkey(model)]
    a = (t[0] * pin + t[1] * pout) / 1e6                      # sweep.py method: cache reads ignored
    b = a + t[2] * pin / 1e6 + t[3] * pin / 1e6               # cache reads/writes at full input price
    c = a + t[2] * CACHED[mkey(model)] / 1e6 + t[3] * pin / 1e6  # cache reads at OpenAI cached-input price
    return a, b, c


print("group | files | episodes | status | model | input | output | cache_read | cache_write | $ sweep-method | $ cache@full | $ cache@cached-rate")
totals = defaultdict(lambda: [0.0, 0.0, 0.0])
model_tot = defaultdict(lambda: defaultdict(lambda: [0, 0, 0, 0]))
for g in sorted(tok):
    ph = g.split()[0]
    for model, t in sorted(tok[g].items()):
        a, b, c = cost(model, t)
        for i, v in enumerate((a, b, c)):
            totals[ph][i] += v
        for i in range(4):
            model_tot[ph][model][i] += t[i]
        print(f"{g} | {nfiles[g]} | {eps[g]} | {','.join(sorted(status[g]))} | {model} | {t[0]} | {t[1]} | {t[2]} | {t[3]} | {a:.3f} | {b:.3f} | {c:.3f}")
print()
for ph in sorted(model_tot):
    for model, t in sorted(model_tot[ph].items()):
        a, b, c = cost(model, t)
        print(f"{ph} total {model}: in {t[0]} out {t[1]} cache_read {t[2]} cache_write {t[3]} | ${a:.2f} / ${b:.2f} / ${c:.2f}")
    print(f"{ph} agents+judge: sweep-method ${totals[ph][0]:.2f}, cache@full ${totals[ph][1]:.2f}, cache@cached-rate ${totals[ph][2]:.2f}")
print("episodes:", {ph: sum(v for g, v in eps.items() if g.startswith(ph)) for ph in ("P3", "P2")},
      "eval files:", {ph: sum(v for g, v in nfiles.items() if g.startswith(ph)) for ph in ("P3", "P2")})

# Proposer: usage is not logged (proposer.py writes system, user, raw; no usage). Estimate from characters.
print("\nproposer cell | calls | in chars | out chars | est in tok | est out tok | $ chars-estimate")
pc = 0.0; ncells = 0
P2_1612 = [P2 / d for d in ("greedy_two", "score_only", "baselines/733_claim_score_only")]
prop_cost = {}
for pl in sorted(list(P3.rglob("proposer_log.jsonl")) + [q for d in P2_1612 for q in d.rglob("proposer_log.jsonl")]):
    ncells += 1
    calls = ic = oc = 0
    for line in pl.open():
        r = json.loads(line)
        calls += 1
        ic += len(r["system"]) + sum(len(u) for u in r["user"])
        oc += len(r["raw"])
    it, ot = ic / CHARS_PER_TOKEN, oc / CHARS_PER_TOKEN
    c = (it * SONNET55[0] + ot * SONNET55[1]) / 1e6
    pc += c
    prop_cost[f"{pl.parent.parts[2]}/{pl.parent.name}"] = c
    print(f"{pl.parent} | {calls} | {ic} | {oc} | {it:.0f} | {ot:.0f} | {c:.4f}")
rej = list(P3.rglob("*rejection*")) + [q for d in P2_1612 for q in d.rglob("*rejection*")]
print(f"proposer cells {ncells}: chars-estimate ${pc:.2f}; flat $0.15/cell ${0.15 * ncells:.2f}; rejection files: {[str(r) for r in rej]}")

# Per Phase 3 cell (seed x ordering x arm): baselines run1+run2 + loop rounds A0-A5 + proposer. Spend-stop flag at $15.
print("\nP3 cell | episodes | $ sweep-method + $0.15 | $ cache@cached-rate + chars | $ cache@full + $0.15 (upper)")
cells = defaultdict(lambda: [0, 0.0, 0.0, 0.0])
for g, models in tok.items():
    if not g.startswith("P3 ") or "check/" in g:
        continue
    ordering, kind, cell = g[3:].split("/")
    k = f"{ordering}/{cell}"
    cells[k][0] += eps[g]
    for model, t in models.items():
        a, b, c = cost(model, t)
        cells[k][1] += a; cells[k][2] += c; cells[k][3] += b
for k in sorted(cells):
    e, a, c, b = cells[k]
    a += 0.15; b += 0.15; c += prop_cost.get(k, 0.0)
    print(f"{k} | {e} | {a:.2f} | {c:.2f} | {b:.2f}{'  spend stop: over $15' if b > 15 else ''}")

# Cross-check only: the rest of results/phase2_claim_monitor (the earlier $16.49 cells), same three pricings, agents+judge only.
rest = [q for q in Path("results/phase2_claim_monitor").rglob("*.eval") if q not in set(files)]
r = [0.0, 0.0, 0.0]; n = 0
for q in rest:
    n += 1
    for model, u in (read_eval_log(str(q), header_only=True).stats.model_usage or {}).items():
        t = [u.input_tokens, u.output_tokens, u.input_tokens_cache_read or 0, u.input_tokens_cache_write or 0]
        for i, v in enumerate(cost(model, t)):
            r[i] += v
print(f"\nP2 earlier cells, {n} eval files, agents+judge: sweep-method ${r[0]:.2f}, cache@cached-rate ${r[2]:.2f}, cache@full ${r[1]:.2f}")
pe = 0.0; ne = 0
for pl in sorted(Path("results/phase2_claim_monitor").rglob("proposer_log.jsonl")):
    if any(pl.is_relative_to(d) for d in P2_1612):
        continue
    ne += 1
    ic = oc = 0
    for line in pl.open():
        r = json.loads(line)
        ic += len(r["system"]) + sum(len(u) for u in r["user"]); oc += len(r["raw"])
    pe += (ic / CHARS_PER_TOKEN * SONNET55[0] + oc / CHARS_PER_TOKEN * SONNET55[1]) / 1e6
print(f"P2 earlier proposer cells {ne}: chars-estimate ${pe:.2f}; flat $0.15/cell ${0.15 * ne:.2f} (neither is in the $16.49)")
```
