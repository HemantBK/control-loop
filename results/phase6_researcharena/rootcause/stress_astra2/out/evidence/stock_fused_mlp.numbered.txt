     1	/*
     2	 * Fused SwiGLU MLP CUDA kernel — optimized implementation
     3	 *
     4	 * Uses cuBLAS (via torch::matmul) for the heavy GEMM work,
     5	 * then fuses the SiLU activation + elementwise multiply in-place.
     6	 */
     7	
     8	#include <torch/extension.h>
     9	#include <cuda_runtime.h>
    10	#include <cuda_fp16.h>
    11	#include <math.h>
    12	
    13	// In-place fused SiLU-multiply: gate[i] = silu(gate[i]) * up[i]
    14	// 4 elements per thread for better ILP
    15	template <typename scalar_t>
    16	__global__ void fused_silu_mul_inplace_kernel(
    17	    scalar_t* __restrict__ gate,         // [N] - modified in place
    18	    const scalar_t* __restrict__ up,     // [N]
    19	    int N
    20	) {
    21	    int base_idx = (blockIdx.x * blockDim.x + threadIdx.x) * 4;
    22	    #pragma unroll
    23	    for (int i = 0; i < 4; i++) {
    24	        int idx = base_idx + i;
    25	        if (idx < N) {
    26	            float g = static_cast<float>(gate[idx]);
    27	            float u = static_cast<float>(up[idx]);
    28	            float silu_g = g / (1.0f + expf(-g));
    29	            gate[idx] = static_cast<scalar_t>(silu_g * u);
    30	        }
    31	    }
    32	}
    33	
    34	// ─── Host wrapper ────────────────────────────────────────────────────────────
    35	
    36	torch::Tensor agent_fused_mlp(
    37	    torch::Tensor input,        // [batch, hidden]
    38	    torch::Tensor gate_weight,  // [intermediate, hidden]
    39	    torch::Tensor up_weight     // [intermediate, hidden]
    40	) {
    41	    // Use torch matmul (cuBLAS) for the GEMMs
    42	    auto gate_proj = torch::mm(input, gate_weight.t());
    43	    auto up_proj = torch::mm(input, up_weight.t());
    44	
    45	    int N = gate_proj.numel();
    46	
    47	    int threads = 256;
    48	    int blocks = (N + threads * 4 - 1) / (threads * 4);
    49	
    50	    AT_DISPATCH_FLOATING_TYPES_AND2(
    51	        at::ScalarType::Half, at::ScalarType::BFloat16,
    52	        input.scalar_type(), "fused_mlp", [&] {
    53	            fused_silu_mul_inplace_kernel<scalar_t><<<blocks, threads>>>(
    54	                gate_proj.data_ptr<scalar_t>(),
    55	                up_proj.data_ptr<scalar_t>(),
    56	                N
    57	            );
    58	        }
    59	    );
    60	
    61	    return gate_proj;  // modified in-place, no extra allocation
    62	}
