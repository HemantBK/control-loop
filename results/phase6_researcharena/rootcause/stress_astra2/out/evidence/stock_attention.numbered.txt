     1	/*
     2	 * Fused Attention CUDA kernel — optimized implementation
     3	 *
     4	 * Computes: output = softmax(Q @ K^T / sqrt(d)) @ V
     5	 * with causal masking support.
     6	 *
     7	 * Optimizations over baseline:
     8	 * - Parallel warp+block reductions instead of serial thread-0 loops
     9	 * - Vectorized dot products where possible
    10	 */
    11	
    12	#include <torch/extension.h>
    13	#include <cuda_runtime.h>
    14	#include <cuda_fp16.h>
    15	#include <math.h>
    16	#include <float.h>
    17	
    18	template <typename scalar_t>
    19	__global__ void attention_kernel(
    20	    const scalar_t* __restrict__ Q,    // [B, H, N_q, D]
    21	    const scalar_t* __restrict__ K,    // [B, H, N_kv, D]
    22	    const scalar_t* __restrict__ V,    // [B, H, N_kv, D]
    23	    scalar_t* __restrict__ output,     // [B, H, N_q, D]
    24	    int q_len,
    25	    int kv_len,
    26	    int head_dim,
    27	    float scale,
    28	    bool causal
    29	) {
    30	    int b = blockIdx.x;
    31	    int h = blockIdx.y;
    32	    int q_pos = blockIdx.z;
    33	
    34	    int num_heads = gridDim.y;
    35	    int q_BH_stride = q_len * head_dim;
    36	    int kv_BH_stride = kv_len * head_dim;
    37	    int base_q = (b * num_heads + h) * q_BH_stride + q_pos * head_dim;
    38	    int full_q_pos = kv_len - q_len + q_pos;
    39	
    40	    extern __shared__ float smem[];
    41	    float* scores = smem;  // [kv_len]
    42	    // Load Q row into shared memory after scores area
    43	    float* s_q = smem + kv_len;  // [head_dim]
    44	
    45	    int tid = threadIdx.x;
    46	    int bdim = blockDim.x;
    47	
    48	    // Cooperatively load Q into shared memory
    49	    for (int d = tid; d < head_dim; d += bdim) {
    50	        s_q[d] = static_cast<float>(Q[base_q + d]);
    51	    }
    52	    __syncthreads();
    53	
    54	    // Step 1: compute attention scores + local max
    55	    float local_max = -FLT_MAX;
    56	    for (int j = tid; j < kv_len; j += bdim) {
    57	        if (causal && j > full_q_pos) {
    58	            scores[j] = -FLT_MAX;
    59	        } else {
    60	            float dot = 0.0f;
    61	            int base_k = (b * num_heads + h) * kv_BH_stride + j * head_dim;
    62	            for (int d = 0; d < head_dim; d++) {
    63	                dot += s_q[d] * static_cast<float>(__ldg(&K[base_k + d]));
    64	            }
    65	            scores[j] = dot * scale;
    66	        }
    67	        local_max = fmaxf(local_max, scores[j]);
    68	    }
    69	
    70	    // Parallel max reduction (warp shuffle + shared memory)
    71	    for (int offset = warpSize / 2; offset > 0; offset >>= 1)
    72	        local_max = fmaxf(local_max, __shfl_down_sync(0xffffffff, local_max, offset));
    73	
    74	    // Inter-warp reduction
    75	    __shared__ float warp_reduce[32];
    76	    int lane = tid & 31;
    77	    int warp_id = tid >> 5;
    78	    int num_warps = (bdim + 31) >> 5;
    79	
    80	    if (lane == 0) warp_reduce[warp_id] = local_max;
    81	    __syncthreads();
    82	
    83	    float global_max;
    84	    if (warp_id == 0) {
    85	        float val = (lane < num_warps) ? warp_reduce[lane] : -FLT_MAX;
    86	        for (int offset = warpSize / 2; offset > 0; offset >>= 1)
    87	            val = fmaxf(val, __shfl_down_sync(0xffffffff, val, offset));
    88	        if (lane == 0) warp_reduce[0] = val;
    89	    }
    90	    __syncthreads();
    91	    global_max = warp_reduce[0];
    92	
    93	    // Step 2: exp and local sum
    94	    float local_sum = 0.0f;
    95	    for (int j = tid; j < kv_len; j += bdim) {
    96	        float e = expf(scores[j] - global_max);
    97	        scores[j] = e;
    98	        local_sum += e;
    99	    }
   100	
   101	    // Parallel sum reduction
   102	    for (int offset = warpSize / 2; offset > 0; offset >>= 1)
   103	        local_sum += __shfl_down_sync(0xffffffff, local_sum, offset);
   104	
   105	    if (lane == 0) warp_reduce[warp_id] = local_sum;
   106	    __syncthreads();
   107	
   108	    float global_sum;
   109	    if (warp_id == 0) {
   110	        float val = (lane < num_warps) ? warp_reduce[lane] : 0.0f;
   111	        for (int offset = warpSize / 2; offset > 0; offset >>= 1)
   112	            val += __shfl_down_sync(0xffffffff, val, offset);
   113	        if (lane == 0) warp_reduce[0] = val;
   114	    }
   115	    __syncthreads();
   116	    global_sum = warp_reduce[0];
   117	
   118	    // Normalize scores
   119	    float inv_sum = 1.0f / global_sum;
   120	    for (int j = tid; j < kv_len; j += bdim) {
   121	        scores[j] *= inv_sum;
   122	    }
   123	    __syncthreads();
   124	
   125	    // Step 3: weighted sum of V
   126	    for (int d = tid; d < head_dim; d += bdim) {
   127	        float acc = 0.0f;
   128	        for (int j = 0; j < kv_len; j++) {
   129	            int base_v = (b * num_heads + h) * kv_BH_stride + j * head_dim;
   130	            acc += scores[j] * static_cast<float>(__ldg(&V[base_v + d]));
   131	        }
   132	        output[base_q + d] = static_cast<scalar_t>(acc);
   133	    }
   134	}

   136	// ─── Host wrapper ────────────────────────────────────────────────────────────
   137	
   138	torch::Tensor agent_attention(
   139	    torch::Tensor Q,      // [batch, heads, q_len, head_dim]
   140	    torch::Tensor K,      // [batch, heads, kv_len, head_dim]
   141	    torch::Tensor V,      // [batch, heads, kv_len, head_dim]
   142	    bool causal
   143	) {
   144	    int batch = Q.size(0);
   145	    int heads = Q.size(1);
   146	    int q_len = Q.size(2);
   147	    int kv_len = K.size(2);
   148	    int head_dim = Q.size(3);
   149	
   150	    float scale = 1.0f / sqrtf(static_cast<float>(head_dim));
   151	    auto output = torch::empty_like(Q);
   152	
   153	    dim3 grid(batch, heads, q_len);
   154	    int threads = min(kv_len, 256);
   155	    int smem_size = (kv_len + head_dim) * sizeof(float);
   156	
   157	    AT_DISPATCH_FLOATING_TYPES_AND2(
   158	        at::ScalarType::Half, at::ScalarType::BFloat16,
   159	        Q.scalar_type(), "attention", [&] {
   160	            attention_kernel<scalar_t><<<grid, threads, smem_size>>>(
   161	                Q.data_ptr<scalar_t>(),
   162	                K.data_ptr<scalar_t>(),
   163	                V.data_ptr<scalar_t>(),
   164	                output.data_ptr<scalar_t>(),
   165	                q_len, kv_len, head_dim, scale, causal
   166	            );
   167	        }
   168	    );
   169	
   170	    return output;
   171	}
